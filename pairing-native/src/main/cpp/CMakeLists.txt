cmake_minimum_required(VERSION 3.22)
project(buddy_pairing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../boringssl boringssl EXCLUDE_FROM_ALL)
if(NOT ANDROID)
    enable_testing()
    add_executable(pairing_tests ../../test/cpp/spake2_test.cpp)
    target_link_libraries(pairing_tests PRIVATE crypto)
    add_test(NAME pairing_protocol COMMAND pairing_tests)
    find_package(JNI REQUIRED)
endif()
add_library(buddy_pairing SHARED pairing.cpp)
target_link_libraries(buddy_pairing PRIVATE crypto)
if(NOT ANDROID)
    target_include_directories(buddy_pairing PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_compile_options(buddy_pairing PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_options(buddy_pairing PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,relro -Wl,-z,now -Wl,--exclude-libs,ALL)
